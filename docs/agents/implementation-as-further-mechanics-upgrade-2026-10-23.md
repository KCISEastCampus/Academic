```python
from fractions import Fraction as F

def closest(r, v, lo=F(0), hi=None):
    q = sum(x*x for x in v)
    if q == 0:
        return None, sum(x*x for x in r)
    t = max(lo, -sum(x*y for x, y in zip(r, v))/q)
    if hi is not None:
        t = min(hi, t)
    return t, sum((x+y*t)**2 for x, y in zip(r, v))

assert closest((F(6), F(8)), (F(-2), F(-2))) == (F(7,2), F(2))
assert closest((F(6), F(8)), (F(-2), F(-2)), hi=F(2)) == (F(2), F(20))
assert closest((F(4), F(3)), (F(2), F(0))) == (F(0), F(25))
assert closest((F(3), F(4)), (F(0), F(0))) == (None, F(25))
assert closest((F(8), F(1)), (F(-2), F(-1)), hi=F(6)) == (F(17,5), F(36,5))
assert closest((F(8), F(1)), (F(-2), F(-1)), hi=F(2)) == (F(2), F(17))
```

from fractions import Fraction as F

def collision(ma, mb, ua, ub, e):
    ma, mb, ua, ub, e = map(F, (ma, mb, ua, ub, e))
    p = ma*ua + mb*ub
    sep = e*(ua-ub)
    va = (p-mb*sep)/(ma+mb)
    vb = va+sep
    loss = (ma*ua**2+mb*ub**2-ma*va**2-mb*vb**2)/2
    assert ma*va+mb*vb == p
    assert vb-va == sep
    assert ma*(va-ua)+mb*(vb-ub) == 0
    assert loss == ma*mb/(ma+mb)*(1-e**2)*(ua-ub)**2/2
    return va, vb, loss

assert collision(2,1,3,-1,F(1,2)) == (F(1),F(3),F(4))
assert collision(1,2,4,-1,F(1,2)) == (F(-1),F(3,2),F(25,4))
assert collision(3,1,2,0,F(1,3)) == (F(4,3),F(2),F(4,3))
assert collision(1,1,2,0,0) == (F(1),F(1),F(1))
assert collision(1,1,4,0,F(1,2)) == (F(1),F(3),F(3))
assert F(1,2)*F(2,5)*15 == 3  # lower peak wall impulse
assert F(1,2)*F(2,5)*30 == 6  # upper peak wall impulse
assert F(6)/2-3 == 0 and F(12)/2-3 == 3  # retry-wall endpoints
assert 2*F(3)**2/2 == 9 and F(3)*(2*3) == 18  # dimensional shortcut
